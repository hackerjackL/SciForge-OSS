% Unified elsarticle template (discipline-agnostic).
% Every problem uses this single document class; elsarticle [preprint,12pt] for drafts.
% Venue-specific adaptation happens at submission time, not during drafting.

\documentclass[preprint,12pt]{elsarticle}

% Math
\usepackage{amsmath,amssymb,amsfonts,amsthm,mathtools}

% Typography
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
% Latin Modern supplies the small math sizes that Computer Modern lacks. With
% the default CM fonts elsarticle emits ten "Font shape ... in size <0.5>/<0.7>/
% <> not available" substitutions plus a size-substitution summary on EVERY
% compile — measured on a minimal frontmatter with no math at all — which makes
% the zero-warning compile requirement unreachable for any paper. lmodern is
% vector Type 1 throughout, so the substitutions disappear and the PDF stays
% fully embedded/searchable.
\usepackage{lmodern}
\usepackage{hyperref}
\usepackage{url}
\usepackage{booktabs}
\usepackage{nicefrac}
\usepackage{microtype}
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{subcaption}
\usepackage{multirow}
\usepackage{algorithm}
\usepackage{algorithmic}

% Float control (v1.4.0 hardening — balanced placement):
% [section]{placeins} keeps every float inside the section that references it
% (no figure drifting pages away); the fraction tuning stops LaTeX from
% promoting a single tall figure to a float-only page or leaving big blanks.
\usepackage{float}
\usepackage[section]{placeins}
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.80}
\renewcommand{\textfraction}{0.15}
\renewcommand{\floatpagefraction}{0.75}

% cleveref must be loaded AFTER hyperref
\usepackage[capitalize,noabbrev]{cleveref}

% elsarticle's \corref/\cnotenum are typesetting-only marks, but hyperref tries
% to flatten the author block into PDF metadata and warns "Token not allowed in
% a PDF string" for each of them on every compile. Strip them from the PDF-string
% context only; the printed corresponding-author mark is unaffected.
\pdfstringdefDisableCommands{%
  \def\corref#1{}%
  \def\cnotenum{}%
  \def\fnref#1{}%
  \def\ead#1{}%
}

% Theorems
\theoremstyle{plain}
\newtheorem{theorem}{Theorem}[section]
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}
\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{assumption}[theorem]{Assumption}
\theoremstyle{remark}
\newtheorem{remark}[theorem]{Remark}

% cleveref names for custom theorem types
\crefname{assumption}{Assumption}{Assumptions}
\Crefname{assumption}{Assumption}{Assumptions}

% Shared math commands
\input{math_commands}

\begin{document}

  % Author block: explicit submission placeholder (never fabricate names, never
  % leak the tool/framework brand into the manuscript — v1.4.0 leak audit).
  \author[1]{[Author list to be completed at submission]\corref{cor1}}
  \cortext[cor1]{Corresponding author.}
  \affiliation[1]{organization={[Affiliation to be completed at submission]}, city={City}, country={Country}}
  \input{sections/0_abstract}
  \begin{keyword}
    keyword one \sep keyword two \sep keyword three
  \end{keyword}
\end{frontmatter}

\input{sections/1_introduction}

\input{sections/4_theory}

\end{document}
